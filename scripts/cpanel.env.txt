# scripts/cpanel.env
#
# مقادیر این فایل مخصوص هاست خودته -- از صفحه‌ی "Setup Python App" تو cPanel
# کپی کن. اونجا یه خط شبیه این نشون می‌ده:
#
#   source /home/USERNAME/virtualenv/personal_hub/3.12/bin/activate && cd /home/USERNAME/personal_hub
#
# از همون خط، دو تا مقدار زیر رو پر کن.
#
# این فایل رو به .gitignore اضافه کن (چون مسیرش مخصوص همین هاسته و نباید
# با ریپازیتوری عمومی گیت‌هاب مخلوط بشه).

# مسیر ریشه‌ی پروژه روی هاست (همون‌جایی که manage.py هست)
APP_ROOT="/home/USERNAME/personal_hub"

# مسیر activate ونوی‌ای که خود cPanel ساخته (نسخه‌ی پایتون رو با نسخه‌ی
# واقعی که تو Setup Python App انتخاب کردی جایگزین کن)
VENV_ACTIVATE="/home/USERNAME/virtualenv/personal_hub/3.12/bin/activate"

# فایلی که Passenger با تاچ‌شدنش، اپ وب (asgi.py) رو ری‌استارت می‌کنه.
# معمولاً همینه؛ اگه تو Setup Python App مسیر متفاوتی دیدی، همینو عوض کن.
PASSENGER_RESTART_FILE="$APP_ROOT/tmp/restart.txt"
